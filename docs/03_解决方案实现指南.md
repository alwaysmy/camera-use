# 解决方案实现指南

## 1. 方案概述

### 1.1 核心思路

使用 `duvc-ctl` 库正确设置 UVC Auto/Manual 曝光标志，解决 OpenCV 图像过暗问题。

### 1.2 技术栈

| 组件 | 用途 |
|------|------|
| duvc-ctl | 封装 DirectShow IAMCameraControl 接口 |
| OpenCV | 视频采集和图像处理 |
| tkinter | GUI 界面 |
| Python 3.11+ | 运行环境 |

### 3.3 打开 duvc-ctl 摄像头

```python
def _open_duvc_camera(self):
    """Open camera via duvc-ctl"""
    if self.duvc_cam:
        try:
            self.duvc_cam.close()
        except:
            pass
    
    try:
        self.duvc_cam = duvc.find_camera('HP')
    except:
        devices = duvc.list_devices()
        if devices:
            self.duvc_cam = duvc.open_camera(devices[0])
```

### 3.4 设置曝光模式

```python
def _set_auto_exposure(self):
    """Set auto-exposure mode"""
    if not self.duvc_cam:
        return
    try:
        self.duvc_cam.set_exposure(0, 'auto')
        self.exposure_mode = "auto"
        print("Auto-exposure enabled")
    except Exception as e:
        print(f"Failed: {e}")

def _set_manual_exposure(self, value):
    """Set manual exposure value"""
    if not self.duvc_cam:
        return
    try:
        self.duvc_cam.set_exposure(int(value), 'manual')
        self.exposure_mode = "manual"
        print(f"Manual exposure: {value}")
    except Exception as e:
        print(f"Failed: {e}")
```

```python
def on_slider_change(self, name, value):
    """Slider value change"""
    val = float(value)
    self.params[name] = val
    
    if self.cap and self.is_previewing:
        if name == "EXPOSURE":
            # 使用 duvc-ctl 控制曝光
            if self.exposure_mode == "manual":
                self._set_manual_exposure(val)
            # Auto模式下忽略滑块
        else:

            if prop_id is not None:
                self.cap.set(prop_id, val)
```

### 4.2 CameraManager 方法

```python
class CameraManager:
    def __init__(self, save_dir=None):
        # ... 其他初始化 ...
        self._duvc_cam = None

    def _get_duvc_camera(self, camera_index=0):
        """获取duvc-ctl摄像头对象"""
        if not HAS_DUVC:
            return None
        try:
            if self._duvc_cam is None:
                self._duvc_cam = duvc.find_camera('HP')
            return self._duvc_cam
        except:
            return None

    def set_auto_exposure(self, camera_index=0):
        """设置自动曝光模式"""
        cam = self._get_duvc_camera(camera_index)
        if cam is None:
            return False
        try:
            cam.set_exposure(0, 'auto')
            return True
        except:
            return False

    def set_manual_exposure(self, value, camera_index=0):
        """设置手动曝光值"""
        cam = self._get_duvc_camera(camera_index)
        if cam is None:
            return False
        try:
            cam.set_exposure(value, 'manual')
            return True
        except:
            return False

    def get_exposure_value(self, camera_index=0):
        """读取当前曝光值"""
        cam = self._get_duvc_camera(camera_index)
        if cam is None:
            return None
        try:
            return cam.get_exposure()
        except:
            return None

    def close_duvc(self):
        """关闭duvc-ctl连接"""
        if self._duvc_cam:
            try:
                self._duvc_cam.close()
            except:
                pass
            self._duvc_cam = None
```

```python
def on_exposure_mode_change(self):
    """Handle exposure mode radio button change"""
    mode = self.exposure_mode_var.get()
    if mode == "auto":
        self._set_auto_exposure()
    else:
        if "EXPOSURE" in self.sliders:
            val = self.sliders["EXPOSURE"][0].get()
            self._set_manual_exposure(val)
```
